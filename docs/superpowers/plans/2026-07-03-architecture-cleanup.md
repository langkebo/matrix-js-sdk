- [ ] **Step 1: Update common/validators.ts to support room aliases**

Add an `allowAlias` option to `validateRoomId`:

```typescript
export function validateRoomId(roomId: string, opts?: { allowAlias?: boolean }): void {
    if (!roomId || typeof roomId !== "string") {
        throw new ValidationError("Room ID is required and must be a string");
    }
    const aliasOk = opts?.allowAlias === true;
    const pattern = aliasOk
        ? /^[!#][a-z0-9._=\-\/]+:[a-z0-9.\-]+$/i
        : /^![a-z0-9._=\-\/]+:[a-z0-9.\-]+$/i;
    if (!pattern.test(roomId)) {
        throw new ValidationError(`Invalid room ID format: ${roomId}`);
    }
}
```

Also add `validateEventType` to `common/validators.ts` (currently only in room-summary-base-manager):

```typescript
export function validateEventType(eventType: string): void {
    if (!eventType || typeof eventType !== "string" || !eventType.includes(".")) {
        throw new ValidationError(`Invalid event type: ${eventType}`);
    }
}
```

```typescript
import { vi } from "vitest";
import type { Transport } from "../../src/managers/base-manager";
import type { Method } from "../../src/http-api/method";
import type { QueryDict, Body } from "../../src/http-api/types";

export interface FakeTransportOptions {
    /** Predefined responses keyed by "METHOD /path" */
    responses?: Record<string, unknown>;
    /** Default response when no match is found */
    defaultResponse?: unknown;
}

export class FakeTransport implements Transport {
    public request = vi.fn<Transport["request"]>();
    private responseMap: Map<string, unknown>;
    private defaultResponse: unknown;

    constructor(opts: FakeTransportOptions = {}) {
        this.responseMap = new Map(Object.entries(opts.responses ?? {}));
        this.defaultResponse = opts.defaultResponse ?? {};

        // Default implementation: lookup response by "METHOD /path"
        this.request.mockImplementation(
            async <T>(
                method: Method,
                path: string,
                _queryParams?: QueryDict,
                _body?: Body,
            ): Promise<T> => {
                const key = `${method} ${path}`;
                const response = this.responseMap.get(key);
                if (response !== undefined) {
                    // Clone to prevent test cross-contamination
                    return structuredClone(response) as T;
                }
                return structuredClone(this.defaultResponse) as T;
            },
        );
    }

    /** Add a response for a specific method+path combination */
    public for(method: Method, path: string): this {
        return this;
    }

    public returning<T>(response: T): this {
        return this;
    }

    /** Reset all mock call history (keeps response map) */
    public resetCalls(): void {
        this.request.mockClear();
    }

    /** Reset everything (call history + response map) */
    public reset(): void {
        this.request.mockClear();
        this.responseMap.clear();
    }
}
```
